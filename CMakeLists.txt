cmake_minimum_required(VERSION 3.20)
project(hyperdsp VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(OBS_INCLUDE_DIR obs-module.h PATH_SUFFIXES obs REQUIRED)
find_library(OBS_LIBRARY NAMES obs REQUIRED)
find_package(Threads REQUIRED)
find_path(LADSPA_INCLUDE_DIR ladspa.h REQUIRED)
add_library(hyperdsp MODULE src/plugin-main.cpp src/mic-filter.cpp src/ladspa-filter.cpp src/native-filter.cpp src/interop/sound_inputs.cpp)
target_include_directories(hyperdsp PRIVATE src "${OBS_INCLUDE_DIR}" "${LADSPA_INCLUDE_DIR}")
target_link_libraries(hyperdsp PRIVATE "${OBS_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(hyperdsp PROPERTIES PREFIX "")
install(TARGETS hyperdsp LIBRARY DESTINATION bin/64bit)
install(FILES LICENSE DESTINATION data)
add_executable(hyperdsp-ladspa-probe tools/ladspa-probe.cpp)
target_include_directories(hyperdsp-ladspa-probe PRIVATE "${LADSPA_INCLUDE_DIR}")
target_link_libraries(hyperdsp-ladspa-probe PRIVATE ${CMAKE_DL_LIBS})
install(TARGETS hyperdsp-ladspa-probe RUNTIME DESTINATION tools)
install(PROGRAMS tools/scan-ladspa.py DESTINATION tools)
include(CTest)
if(BUILD_TESTING)
    add_executable(verify-modulation-matrix tests/verify-modulation-matrix.cpp)
    target_include_directories(verify-modulation-matrix PRIVATE src)
    add_test(NAME modulation-matrix COMMAND verify-modulation-matrix)
    add_executable(verify-reactive-source tests/verify-reactive-source.cpp)
    target_include_directories(verify-reactive-source PRIVATE src)
    add_test(NAME reactive-source COMMAND verify-reactive-source)
    add_executable(verify-band-energy tests/verify-band-energy.cpp)
    target_include_directories(verify-band-energy PRIVATE src)
    add_test(NAME band-energy COMMAND verify-band-energy)
    add_executable(verify-native-filter tests/verify-native-filter.cpp)
    target_include_directories(verify-native-filter PRIVATE src)
    add_test(NAME native-filter COMMAND verify-native-filter)
    add_executable(verify-native-obs tests/verify-native-obs.cpp)
    target_include_directories(verify-native-obs PRIVATE "${OBS_INCLUDE_DIR}")
    target_link_libraries(verify-native-obs PRIVATE "${OBS_LIBRARY}" Threads::Threads)
    add_test(NAME native-obs COMMAND verify-native-obs "$<TARGET_FILE:hyperdsp>" "${CMAKE_CURRENT_SOURCE_DIR}")
    set_tests_properties(native-obs PROPERTIES TIMEOUT 30)
    add_executable(verify-modulation tests/verify-modulation.cpp)
    target_include_directories(verify-modulation PRIVATE src)
    add_test(NAME modulation COMMAND verify-modulation)
    add_executable(verify-effect-mix tests/verify-effect-mix.cpp)
    target_include_directories(verify-effect-mix PRIVATE src)
    add_test(NAME effect-mix COMMAND verify-effect-mix)
    add_executable(verify-mic-engine tests/verify-mic-dsp-engine.cpp)
    target_include_directories(verify-mic-engine PRIVATE src)
    target_compile_options(verify-mic-engine PRIVATE -UNDEBUG)
    add_test(NAME mic-engine COMMAND verify-mic-engine)
    add_executable(verify-mic-obs tests/verify-mic-obs-audio.cpp)
    target_include_directories(verify-mic-obs PRIVATE "${OBS_INCLUDE_DIR}")
    target_link_libraries(verify-mic-obs PRIVATE "${OBS_LIBRARY}" Threads::Threads)
    add_test(NAME mic-obs COMMAND verify-mic-obs "$<TARGET_FILE:hyperdsp>" "${CMAKE_CURRENT_SOURCE_DIR}")
    set_tests_properties(mic-obs PROPERTIES TIMEOUT 30)
    add_executable(verify-ladspa-obs tests/verify-ladspa-obs.cpp)
    target_include_directories(verify-ladspa-obs PRIVATE "${OBS_INCLUDE_DIR}")
    target_link_libraries(verify-ladspa-obs PRIVATE "${OBS_LIBRARY}" Threads::Threads)
    find_file(LADSPA_TEST_PLUGIN amp.so PATHS /usr/lib/ladspa /usr/local/lib/ladspa /usr/lib/x86_64-linux-gnu/ladspa)
    if(LADSPA_TEST_PLUGIN)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME ladspa-catalog-scan COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/scan-ladspa.py"
            --probe "$<TARGET_FILE:hyperdsp-ladspa-probe>" --output "${CMAKE_CURRENT_BINARY_DIR}/test-catalog.json" "${LADSPA_TEST_PLUGIN}")
        set_tests_properties(ladspa-catalog-scan PROPERTIES FIXTURES_SETUP ladspa_catalog TIMEOUT 15)
        add_test(NAME ladspa-obs COMMAND verify-ladspa-obs "$<TARGET_FILE:hyperdsp>" "${CMAKE_CURRENT_SOURCE_DIR}" "${LADSPA_TEST_PLUGIN}")
        set_tests_properties(ladspa-obs PROPERTIES TIMEOUT 30 FIXTURES_REQUIRED ladspa_catalog ENVIRONMENT "HYPERDSP_CATALOG=${CMAKE_CURRENT_BINARY_DIR}/test-catalog.json")
    endif()
endif()
